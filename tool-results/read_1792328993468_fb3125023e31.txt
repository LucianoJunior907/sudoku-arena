     1→     1→"use client";
     2→     2→

     4→     4→import { useAuth } from "@/lib/supabase/auth";
     5→     5→import { useI18n } from "@/lib/i18n/context";
     6→     6→import { Button } from "@/components/ui/button";
     7→     7→import { Input } from "@/components/ui/input";
     8→     8→import { Label } from "@/components/ui/label";

    10→    10→import { toast } from "sonner";
    11→    11→import { checkUsernameAvailable, setUsername } from "@/lib/supabase/matches";
    12→    12→import { isSupabaseConfigured } from "@/lib/supabase/client";
    13→    13→import { cn } from "@/lib/utils";
    14→    14→
    15→    15→export function AuthDialog() {
    16→    16→  const { locale } = useI18n();
    17→    17→  const { user, loading, isGuest, signInWithUsernameOrEmail, signUpWithEmail, signInWithGoogle, signOut, enterGuestMode, exitGuestMode } = useAuth();
    18→    18→  const [mode, setMode] = useState<"signin" | "signup">("signin");
    19→    19→  const [email, setEmail] = useState("");
    20→    20→  const [password, setPassword] = useState("");
    21→    21→  const [confirmPassword, setConfirmPassword] = useState("");
    22→    22→  const [name, setName] = useState("");
    23→    23→  const [username, setUsernameState] = useState("");
    24→    24→  const [usernameStatus, setUsernameStatus] = useState<"idle" | "checking" | "available" | "taken" | "invalid">("idle");
    25→    25→  const [loginIdentifier, setLoginIdentifier] = useState("");
    26→    26→  const [busy, setBusy] = useState(false);
    27→    27→  const [showPassword, setShowPassword] = useState(false);
    28→    28→  const [showConfirm, setShowConfirm] = useState(false);
    29→    29→
    30→    30→  const checkUsername = async (value: string) => {
    31→    31→    setUsernameState(value);
    32→    32→    if (value.length < 3) {
    33→    33→      setUsernameStatus("invalid");
    34→    34→      return;
    35→    35→    }
    36→    36→    setUsernameStatus("checking");
    37→    37→    const result = await checkUsernameAvailable(value);
    38→    38→    if (result.available) {
    39→    39→      setUsernameStatus("available");
    40→    40→    } else if (result.error) {
    41→    41→      // Distinguish validation errors (format/length) from "actually taken"
    42→    42→      const err = result.error.toLowerCase();
    43→    43→      if (err.includes("character") || err.includes("must be") || err.includes("only letters")) {
    44→    44→        setUsernameStatus("invalid");
    45→    45→      } else if (err.includes("taken") || err.includes("already") || err.includes("exists")) {
    46→    46→        setUsernameStatus("taken");
    47→    47→      } else {
    48→    48→        // RPC/permission/network error — don't block the user, allow them to try
    49→    49→        // (the final re-check on submit will catch real conflicts)
    50→    50→        setUsernameStatus("available");
    51→    51→      }
    52→    52→    } else {
    53→    53→      // No error but not available = genuinely taken
    54→    54→      setUsernameStatus("taken");
    55→    55→    }
    56→    56→  };
    57→    57→
    58→    58→  const handleSubmit = async (e: React.FormEvent) => {
    59→    59→    e.preventDefault();
    60→    60→    setBusy(true);
    61→    61→
    62→    62→    if (mode === "signup") {
    63→    63→      // Validate username
    64→    64→      if (username.trim().length < 3) {
    65→    65→        toast.error(locale === "pt-BR" ? "Nome de usuário deve ter no mínimo 3 caracteres." : "Username must be at least 3 characters.");
    66→    66→        setBusy(false);
    67→    67→        return;
    68→    68→      }
    69→    69→      if (usernameStatus === "taken") {
    70→    70→        toast.error(locale === "pt-BR" ? "Nome de usuário já está em uso." : "Username already taken.");
    71→    71→        setBusy(false);
    72→    72→        return;
    73→    73→      }
    74→    74→      // Validate password confirmation
    75→    75→      if (password !== confirmPassword) {
    76→    76→        toast.error(locale === "pt-BR" ? "As senhas não coincidem." : "Passwords do not match.");
    77→    77→        setBusy(false);
    78→    78→        return;
    79→    79→      }
    80→    80→      // Re-check availability (but don't block on RPC errors — set_username will validate)
    81→    81→      const check = await checkUsernameAvailable(username);
    82→    82→      if (!check.available && (!check.error || check.error.toLowerCase().includes("taken") || check.error.toLowerCase().includes("already"))) {
    83→    83→        toast.error(locale === "pt-BR" ? "Nome de usuário já está em uso." : "Username already taken.");
    84→    84→        setUsernameStatus("taken");
    85→    85→        setBusy(false);
    86→    86→        return;
    87→    87→      }
    88→    88→

    91→    91→        toast.error(error.message);

    94→    94→        setTimeout(async () => {
    95→    95→          const result = await setUsername(username);
    96→    96→          if (!result.success) {
    97→    97→            toast.error(result.error || "Failed to set username");
    98→    98→          }
    99→    99→        }, 1500);
   100→   100→        toast.success(locale === "pt-BR" ? "Conta criada!" : "Account created!");
   101→   101→      }
   102→   102→    } else {
   103→   103→      // Login by username or email
   104→   104→      if (!loginIdentifier.trim()) {
   105→   105→        toast.error(locale === "pt-BR" ? "Digite seu nome de usuário ou email." : "Enter your username or email.");
   106→   106→        setBusy(false);
   107→   107→        return;
   108→   108→      }
   109→   109→      const { error } = await signInWithUsernameOrEmail(loginIdentifier, password);
   110→   110→      if (error) {
   111→   111→        toast.error(locale === "pt-BR" ? "Credenciais inválidas." : "Invalid credentials.");
   112→   112→      } else {
   113→   113→        toast.success(locale === "pt-BR" ? "Bem-vindo!" : "Welcome!");
   114→   114→      }
   115→   115→    }
   116→   116→    setBusy(false);
   117→   117→  };
   118→   118→
   119→   119→  const handleGoogle = async () => {
   120→   120→    setBusy(true);
   121→   121→    const { error } = await signInWithGoogle();
   122→   122→    if (error) toast.error(error.message);
   123→   123→    setBusy(false);
   124→   124→  };
   125→   125→
   126→   126→  // If user is signed in (or in guest mode), show profile card + logout button
   127→   127→  if (user) {
   128→   128→    return (
   129→   129→      <LoggedInHeader
   130→   130→        displayName={user.displayName}
   131→   131→        onSignOut={signOut}
   132→   132→        locale={locale}
   133→   133→        isGuest={isGuest}
   134→   134→        onLogin={exitGuestMode}
   135→   135→      />
   136→   136→    );
   137→   137→  }
   138→   138→
   139→   139→  if (loading) {
   140→   140→    return (
   141→   141→      <Button variant="ghost" size="sm" disabled>
   142→   142→        <Loader2 className="h-4 w-4 animate-spin" />
   143→   143→      </Button>
   144→   144→    );
   145→   145→  }
   146→   146→
   147→   147→  // Full-screen opaque login overlay
   148→   148→  return (
   149→   149→    <div className="fixed inset-0 z-[200] flex items-center justify-center bg-background overflow-hidden">
   150→   150→      {/* Background pattern */}
   151→   151→      <div className="absolute inset-0 bg-arena-grid opacity-30" />
   152→   152→      <div className="absolute -right-20 -top-20 h-60 w-60 rounded-full bg-primary/10 blur-3xl" />
   153→   153→      <div className="absolute -bottom-20 left-1/3 h-60 w-60 rounded-full bg-[var(--warning)]/10 blur-3xl" />
   154→   154→
   155→   155→      {/* Grid-aligned numbers */}
   156→   156→      <GridNumbers />
   157→   157→
   158→   158→      <div className="relative w-full max-w-md p-6">
   159→   159→        <div className="mb-6 text-center">
   160→   160→          <div className="mx-auto mb-4 flex h-16 w-16 items-center justify-center rounded-2xl bg-primary/10 ring-1 ring-primary/20">
   161→   161→            <Grid3x3 className="h-8 w-8 text-primary" />
   162→   162→          </div>
   163→   163→          <h1 className="text-2xl font-bold">Sudoku Arena</h1>
   164→   164→          <p className="mt-1 text-sm text-muted-foreground">
   165→   165→            {locale === "pt-BR" ? "Treine. Compita. Evolua." : "Train. Compete. Evolve."}
   166→   166→          </p>
   167→   167→        </div>
   168→   168→
   169→   169→        <div className="rounded-xl border border-border bg-card p-6 shadow-xl">
   170→   170→          <h2 className="mb-1 text-center text-lg font-semibold">
   171→   171→            {mode === "signin"
   172→   172→              ? (locale === "pt-BR" ? "Entrar" : "Sign In")
   173→   173→              : (locale === "pt-BR" ? "Criar conta" : "Sign Up")}

   175→   175→          <p className="mb-4 text-center text-xs text-muted-foreground">
   176→   176→            {locale === "pt-BR"

   178→   178→              : "Sign in to compete online, sync your progress, and play with friends."}
   179→   179→          </p>
   180→   180→
   181→   181→          {/* Supabase not configured warning (e.g. missing env vars on Vercel) */}
   182→   182→          {!isSupabaseConfigured && (
   183→   183→            <div className="mb-4 flex items-start gap-2 rounded-lg border border-amber-500/40 bg-amber-500/10 p-3 text-xs">
   184→   184→              <AlertTriangle className="mt-0.5 h-4 w-4 shrink-0 text-amber-600 dark:text-amber-400" />
   185→   185→              <div className="text-amber-700 dark:text-amber-300">
   186→   186→                <p className="font-semibold">
   187→   187→                  {locale === "pt-BR" ? "Backend não configurado" : "Backend not configured"}
   188→   188→                </p>
   189→   189→                <p className="mt-0.5 leading-relaxed">
   190→   190→                  {locale === "pt-BR"
   191→   191→                    ? "As variáveis do Supabase não foram definidas. Login não funcionará, mas você pode jogar como visitante."
   192→   192→                    : "Supabase environment variables are missing. Login won't work, but you can play as guest."}
   193→   193→                </p>
   194→   194→              </div>
   195→   195→            </div>
   196→   196→          )}
   197→   197→
   198→   198→          {/* Google OAuth */}
   199→   199→          <Button variant="outline" className="w-full mb-3" onClick={handleGoogle} disabled={busy || !isSupabaseConfigured}>
   200→   200→            <Chrome className="mr-2 h-4 w-4" />
   201→   201→            {locale === "pt-BR" ? "Continuar com Google" : "Continue with Google"}
   202→   202→          </Button>
   203→   203→
   204→   204→          <div className="flex items-center gap-2 mb-3">
   205→   205→            <div className="h-px flex-1 bg-border" />
   206→   206→            <span className="text-xs text-muted-foreground">{locale === "pt-BR" ? "ou" : "or"}</span>
   207→   207→            <div className="h-px flex-1 bg-border" />
   208→   208→          </div>
   209→   209→
   210→   210→          {/* Form */}
   211→   211→          <form onSubmit={handleSubmit} className="space-y-3">
   212→   212→            {mode === "signin" ? (
   213→   213→              /* Login by username or email */
   214→   214→              <div className="space-y-1.5">
   215→   215→                <Label htmlFor="loginId">
   216→   216→                  {locale === "pt-BR" ? "Nome de usuário ou email" : "Username or email"}
   217→   217→                </Label>
   218→   218→                <div className="relative">
   219→   219→                  <AtSign className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   220→   220→                  <Input
   221→   221→                    id="loginId"
   222→   222→                    value={loginIdentifier}
   223→   223→                    onChange={(e) => setLoginIdentifier(e.target.value)}
   224→   224→                    placeholder={locale === "pt-BR" ? "@usuario ou email" : "@username or email"}
   225→   225→                    className="pl-9"
   226→   226→                    required
   227→   227→                  />
   228→   228→                </div>
   229→   229→              </div>
   230→   230→            ) : (
   231→   231→              /* Signup fields */
   232→   232→              <>
   233→   233→                <div className="space-y-1.5">
   234→   234→                  <Label htmlFor="name">{locale === "pt-BR" ? "Nome" : "Name"}</Label>
   235→   235→                  <div className="relative">
   236→   236→                    <User className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   237→   237→                    <Input
   238→   238→                      id="name"
   239→   239→                      value={name}
   240→   240→                      onChange={(e) => setName(e.target.value)}
   241→   241→                      placeholder={locale === "pt-BR" ? "Seu nome" : "Your name"}
   242→   242→                      className="pl-9"
   243→   243→                    />
   244→   244→                  </div>
   245→   245→                </div>
   246→   246→                <div className="space-y-1.5">
   247→   247→                  <Label htmlFor="username">
   248→   248→                    {locale === "pt-BR" ? "Nome de usuário" : "Username"}
   249→   249→                  </Label>
   250→   250→                  <div className="relative">
   251→   251→                    <AtSign className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   252→   252→                    <Input
   253→   253→                      id="username"
   254→   254→                      value={username}
   255→   255→                      onChange={(e) => {
   256→   256→                        const val = e.target.value.replace(/[^a-zA-Z0-9_]/g, "");
   257→   257→                        setUsernameState(val);
   258→   258→                        if (val.length >= 3) {
   259→   259→                          checkUsername(val);
   260→   260→                        } else {
   261→   261→                          setUsernameStatus("invalid");
   262→   262→                        }
   263→   263→                      }}
   264→   264→                      placeholder="username"
   265→   265→                      className={cn(
   266→   266→                        "pl-9",
   267→   267→                        usernameStatus === "available" && "border-[var(--success)] focus-visible:ring-[var(--success)]",
   268→   268→                        usernameStatus === "taken" && "border-[var(--destructive)] focus-visible:ring-[var(--destructive)]",
   269→   269→                      )}
   270→   270→                      maxLength={20}
   271→   271→                    />
   272→   272→                    {usernameStatus === "checking" && (
   273→   273→                      <Loader2 className="absolute right-3 top-1/2 h-4 w-4 -translate-y-1/2 animate-spin text-muted-foreground" />
   274→   274→                    )}
   275→   275→                    {usernameStatus === "available" && (
   276→   276→                      <span className="absolute right-3 top-1/2 -translate-y-1/2 text-xs font-medium text-[var(--success)]">
   277→   277→                        {locale === "pt-BR" ? "Disponível" : "Available"}
   278→   278→                      </span>
   279→   279→                    )}
   280→   280→                    {usernameStatus === "taken" && (
   281→   281→                      <span className="absolute right-3 top-1/2 -translate-y-1/2 text-xs font-medium text-[var(--destructive)]">
   282→   282→                        {locale === "pt-BR" ? "Em uso" : "Taken"}
   283→   283→                      </span>
   284→   284→                    )}
   285→   285→                  </div>
   286→   286→                  {usernameStatus === "invalid" && username.length > 0 && (
   287→   287→                    <p className="text-xs text-muted-foreground">
   288→   288→                      {locale === "pt-BR" ? "Mínimo 3 caracteres. Apenas letras, números e _." : "Min 3 characters. Letters, numbers and _ only."}
   289→   289→                    </p>
   290→   290→                  )}
   291→   291→                </div>
   292→   292→                <div className="space-y-1.5">
   293→   293→                  <Label htmlFor="email">Email</Label>
   294→   294→                  <div className="relative">
   295→   295→                    <Mail className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   296→   296→                    <Input
   297→   297→                      id="email"
   298→   298→                      type="email"
   299→   299→                      value={email}
   300→   300→                      onChange={(e) => setEmail(e.target.value)}
   301→   301→                      placeholder="email@example.com"
   302→   302→                      className="pl-9"
   303→   303→                      required
   304→   304→                    />
   305→   305→                  </div>
   306→   306→                </div>
   307→   307→              </>
   308→   308→            )}
   309→   309→
   310→   310→            {/* Password */}
   311→   311→            <div className="space-y-1.5">
   312→   312→              <Label htmlFor="password">{locale === "pt-BR" ? "Senha" : "Password"}</Label>
   313→   313→              <div className="relative">
   314→   314→                <Lock className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   315→   315→                <Input
   316→   316→                  id="password"
   317→   317→                  type={showPassword ? "text" : "password"}
   318→   318→                  value={password}
   319→   319→                  onChange={(e) => setPassword(e.target.value)}
   320→   320→                  placeholder="••••••••"
   321→   321→                  className="pl-9 pr-9"
   322→   322→                  required

   324→   324→                />
   325→   325→                <button

   331→   331→                </button>
   332→   332→              </div>
   333→   333→            </div>
   334→   334→
   335→   335→            {/* Password confirmation (signup only) */}
   336→   336→            {mode === "signup" && (
   337→   337→              <div className="space-y-1.5">
   338→   338→                <Label htmlFor="confirmPassword">
   339→   339→                  {locale === "pt-BR" ? "Confirmar senha" : "Confirm password"}
   340→   340→                </Label>
   341→   341→                <div className="relative">
   342→   342→                  <Lock className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   343→   343→                  <Input

   345→   345→                    type={showConfirm ? "text" : "password"}

   540→   540→function GridNumbers() {
   541→   541→  const [cells, setCells] = useState<GridCell[]>(() => generateGridCells());
   542→   542→
   543→   543→  useEffect(() => {
   544→   544→    const interval = setInterval(() => {
   545→   545→      setCells((prev) =>
   546→   546→        prev.map((c) => {
   547→   547→          if (c.phase === "hidden") {
   548→   548→            if (Math.random() < 0.2) {
   549→   549→              return {
   550→   550→                ...c,
   551→   551→                digit: Math.floor(Math.random() * 9) + 1,
   552→   552→                phase: "in" as const,
   553→   553→              };
   554→   554→            }
   555→   555→            return c;
   556→   556→          }
   557→   557→          if (c.phase === "in") {
   558→   558→            if (c.opacity >= 0.12) return { ...c, phase: "hold" as const };
   559→   559→            return { ...c, opacity: c.opacity + 0.02 };
   560→   560→          }
   561→   561→          if (c.phase === "hold") {
   562→   562→            if (c.holdTime <= 0) return { ...c, phase: "out" as const };
   563→   563→            return { ...c, holdTime: c.holdTime - 1 };
   564→   564→          }
   565→   565→          if (c.phase === "out") {
   566→   566→            if (c.opacity <= 0) return { ...c, phase: "hidden" as const, opacity: 0 };
   567→   567→            return { ...c, opacity: c.opacity - 0.02 };
   568→   568→          }
   569→   569→          return c;
   570→   570→        })
   571→   571→      );
   572→   572→    }, 100);
   573→   573→    return () => clearInterval(interval);
   574→   574→  }, []);
   575→   575→
   576→   576→  return (
   577→   577→    <div className="pointer-events-none absolute inset-0">
   578→   578→      {cells.map((c) => (
   579→   579→        <span
   580→   580→          key={c.id}
   581→   581→          className="absolute flex items-center justify-center font-bold tabular-nums text-muted-foreground transition-opacity duration-100"
   582→   582→          style={{
   583→   583→            left: `${c.col * 32}px`,
   584→   584→            top: `${c.row * 32}px`,
   585→   585→            width: "32px",
   586→   586→            height: "32px",
   587→   587→            fontSize: "1rem",
   588→   588→            opacity: c.phase === "hidden" ? 0 : c.opacity,
   589→   589→          }}
   590→   590→        >
   591→   591→          {c.phase !== "hidden" ? c.digit : ""}
   592→   592→        </span>
   593→   593→      ))}
   594→   594→    </div>
   595→   595→  );
   596→   596→}
