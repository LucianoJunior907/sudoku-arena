     1→"use client";
     2→
     3→import { useState, useEffect } from "react";
     4→import { useAuth } from "@/lib/supabase/auth";
     5→import { useI18n } from "@/lib/i18n/context";
     6→import { Button } from "@/components/ui/button";
     7→import { Input } from "@/components/ui/input";
     8→import { Label } from "@/components/ui/label";
     9→import { Grid3x3, Mail, Lock, User, Chrome, LogOut, Loader2, AtSign, Eye, EyeOff, Ghost, LogIn, AlertTriangle } from "lucide-react";
    10→import { toast } from "sonner";
    11→import { checkUsernameAvailable, setUsername } from "@/lib/supabase/matches";
    12→import { isSupabaseConfigured } from "@/lib/supabase/client";
    13→import { cn } from "@/lib/utils";
    14→
    15→export function AuthDialog() {
    16→  const { locale } = useI18n();
    17→  const { user, loading, isGuest, signInWithUsernameOrEmail, signUpWithEmail, signInWithGoogle, signOut, enterGuestMode, exitGuestMode } = useAuth();
    18→  const [mode, setMode] = useState<"signin" | "signup">("signin");
    19→  const [email, setEmail] = useState("");
    20→  const [password, setPassword] = useState("");
    21→  const [confirmPassword, setConfirmPassword] = useState("");
    22→  const [name, setName] = useState("");
    23→  const [username, setUsernameState] = useState("");
    24→  const [usernameStatus, setUsernameStatus] = useState<"idle" | "checking" | "available" | "taken" | "invalid">("idle");
    25→  const [loginIdentifier, setLoginIdentifier] = useState("");
    26→  const [busy, setBusy] = useState(false);
    27→  const [showPassword, setShowPassword] = useState(false);
    28→  const [showConfirm, setShowConfirm] = useState(false);
    29→
    30→  const checkUsername = async (value: string) => {
    31→    setUsernameState(value);
    32→    if (value.length < 3) {
    33→      setUsernameStatus("invalid");
    34→      return;
    35→    }
    36→    setUsernameStatus("checking");
    37→    const result = await checkUsernameAvailable(value);
    38→    if (result.available) {
    39→      setUsernameStatus("available");
    40→    } else if (result.error) {
    41→      // Distinguish validation errors (format/length) from "actually taken"
    42→      const err = result.error.toLowerCase();
    43→      if (err.includes("character") || err.includes("must be") || err.includes("only letters")) {
    44→        setUsernameStatus("invalid");
    45→      } else if (err.includes("taken") || err.includes("already") || err.includes("exists")) {
    46→        setUsernameStatus("taken");
    47→      } else {
    48→        // RPC/permission/network error — don't block the user, allow them to try
    49→        // (the final re-check on submit will catch real conflicts)
    50→        setUsernameStatus("available");
    51→      }
    52→    } else {
    53→      // No error but not available = genuinely taken
    54→      setUsernameStatus("taken");
    55→    }
    56→  };
    57→
    58→  const handleSubmit = async (e: React.FormEvent) => {
    59→    e.preventDefault();
    60→    setBusy(true);
    61→
    62→    if (mode === "signup") {
    63→      // Validate username
    64→      if (username.trim().length < 3) {
    65→        toast.error(locale === "pt-BR" ? "Nome de usuário deve ter no mínimo 3 caracteres." : "Username must be at least 3 characters.");
    66→        setBusy(false);
    67→        return;
    68→      }
    69→      if (usernameStatus === "taken") {
    70→        toast.error(locale === "pt-BR" ? "Nome de usuário já está em uso." : "Username already taken.");
    71→        setBusy(false);
    72→        return;
    73→      }
    74→      // Validate password confirmation
    75→      if (password !== confirmPassword) {
    76→        toast.error(locale === "pt-BR" ? "As senhas não coincidem." : "Passwords do not match.");
    77→        setBusy(false);
    78→        return;
    79→      }
    80→      // Re-check availability (but don't block on RPC errors — set_username will validate)
    81→      const check = await checkUsernameAvailable(username);
    82→      if (!check.available && (!check.error || check.error.toLowerCase().includes("taken") || check.error.toLowerCase().includes("already"))) {
    83→        toast.error(locale === "pt-BR" ? "Nome de usuário já está em uso." : "Username already taken.");
    84→        setUsernameStatus("taken");
    85→        setBusy(false);
    86→        return;
    87→      }
    88→
    89→      const { error } = await signUpWithEmail(email, password, name || username, username);
    90→      if (error) {
    91→        toast.error(error.message);
    92→      } else {
    93→        // After signup, set the username in the profile
    94→        setTimeout(async () => {
    95→          const result = await setUsername(username);
    96→          if (!result.success) {
    97→            toast.error(result.error || "Failed to set username");
    98→          }
    99→        }, 1500);
   100→        toast.success(locale === "pt-BR" ? "Conta criada!" : "Account created!");
   101→      }
   102→    } else {
   103→      // Login by username or email
   104→      if (!loginIdentifier.trim()) {
   105→        toast.error(locale === "pt-BR" ? "Digite seu nome de usuário ou email." : "Enter your username or email.");
   106→        setBusy(false);
   107→        return;
   108→      }
   109→      const { error } = await signInWithUsernameOrEmail(loginIdentifier, password);
   110→      if (error) {
   111→        toast.error(locale === "pt-BR" ? "Credenciais inválidas." : "Invalid credentials.");
   112→      } else {
   113→        toast.success(locale === "pt-BR" ? "Bem-vindo!" : "Welcome!");
   114→      }
   115→    }
   116→    setBusy(false);
   117→  };
   118→
   119→  const handleGoogle = async () => {
   120→    setBusy(true);
   121→    const { error } = await signInWithGoogle();
   122→    if (error) toast.error(error.message);
   123→    setBusy(false);
   124→  };
   125→
   126→  // If user is signed in (or in guest mode), show profile card + logout button
   127→  if (user) {
   128→    return (
   129→      <LoggedInHeader
   130→        displayName={user.displayName}
   131→        onSignOut={signOut}
   132→        locale={locale}
   133→        isGuest={isGuest}
   134→        onLogin={exitGuestMode}
   135→      />
   136→    );
   137→  }
   138→
   139→  if (loading) {
   140→    return (
   141→      <Button variant="ghost" size="sm" disabled>
   142→        <Loader2 className="h-4 w-4 animate-spin" />
   143→      </Button>
   144→    );
   145→  }
   146→
   147→  // Full-screen opaque login overlay
   148→  return (
   149→    <div className="fixed inset-0 z-[200] flex items-center justify-center bg-background overflow-hidden">
   150→      {/* Background pattern */}
   151→      <div className="absolute inset-0 bg-arena-grid opacity-30" />
   152→      <div className="absolute -right-20 -top-20 h-60 w-60 rounded-full bg-primary/10 blur-3xl" />
   153→      <div className="absolute -bottom-20 left-1/3 h-60 w-60 rounded-full bg-[var(--warning)]/10 blur-3xl" />
   154→
   155→      {/* Grid-aligned numbers */}
   156→      <GridNumbers />
   157→
   158→      <div className="relative w-full max-w-md p-6">
   159→        <div className="mb-6 text-center">
   160→          <div className="mx-auto mb-4 flex h-16 w-16 items-center justify-center rounded-2xl bg-primary/10 ring-1 ring-primary/20">
   161→            <Grid3x3 className="h-8 w-8 text-primary" />
   162→          </div>
   163→          <h1 className="text-2xl font-bold">Sudoku Arena</h1>
   164→          <p className="mt-1 text-sm text-muted-foreground">
   165→            {locale === "pt-BR" ? "Treine. Compita. Evolua." : "Train. Compete. Evolve."}
   166→          </p>
   167→        </div>
   168→
   169→        <div className="rounded-xl border border-border bg-card p-6 shadow-xl">
   170→          <h2 className="mb-1 text-center text-lg font-semibold">
   171→            {mode === "signin"
   172→              ? (locale === "pt-BR" ? "Entrar" : "Sign In")
   173→              : (locale === "pt-BR" ? "Criar conta" : "Sign Up")}
   174→          </h2>
   175→          <p className="mb-4 text-center text-xs text-muted-foreground">
   176→            {locale === "pt-BR"
   177→              ? "Entre para competir online, salvar seu progresso e jogar com amigos."
   178→              : "Sign in to compete online, sync your progress, and play with friends."}
   179→          </p>
   180→
   181→          {/* Supabase not configured warning (e.g. missing env vars on Vercel) */}
   182→          {!isSupabaseConfigured && (
   183→            <div className="mb-4 flex items-start gap-2 rounded-lg border border-amber-500/40 bg-amber-500/10 p-3 text-xs">
   184→              <AlertTriangle className="mt-0.5 h-4 w-4 shrink-0 text-amber-600 dark:text-amber-400" />
   185→              <div className="text-amber-700 dark:text-amber-300">
   186→                <p className="font-semibold">
   187→                  {locale === "pt-BR" ? "Backend não configurado" : "Backend not configured"}
   188→                </p>
   189→                <p className="mt-0.5 leading-relaxed">
   190→                  {locale === "pt-BR"
   191→                    ? "As variáveis do Supabase não foram definidas. Login não funcionará, mas você pode jogar como visitante."
   192→                    : "Supabase environment variables are missing. Login won't work, but you can play as guest."}
   193→                </p>
   194→              </div>
   195→            </div>
   196→          )}
   197→
   198→          {/* Google OAuth */}
   199→          <Button variant="outline" className="w-full mb-3" onClick={handleGoogle} disabled={busy || !isSupabaseConfigured}>
   200→            <Chrome className="mr-2 h-4 w-4" />
   201→            {locale === "pt-BR" ? "Continuar com Google" : "Continue with Google"}
   202→          </Button>
   203→
   204→          <div className="flex items-center gap-2 mb-3">
   205→            <div className="h-px flex-1 bg-border" />
   206→            <span className="text-xs text-muted-foreground">{locale === "pt-BR" ? "ou" : "or"}</span>
   207→            <div className="h-px flex-1 bg-border" />
   208→          </div>
   209→
   210→          {/* Form */}
   211→          <form onSubmit={handleSubmit} className="space-y-3">
   212→            {mode === "signin" ? (
   213→              /* Login by username or email */
   214→              <div className="space-y-1.5">
   215→                <Label htmlFor="loginId">
   216→                  {locale === "pt-BR" ? "Nome de usuário ou email" : "Username or email"}
   217→                </Label>
   218→                <div className="relative">
   219→                  <AtSign className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   220→                  <Input
   221→                    id="loginId"
   222→                    value={loginIdentifier}
   223→                    onChange={(e) => setLoginIdentifier(e.target.value)}
   224→                    placeholder={locale === "pt-BR" ? "@usuario ou email" : "@username or email"}
   225→                    className="pl-9"
   226→                    required
   227→                  />
   228→                </div>
   229→              </div>
   230→            ) : (
   231→              /* Signup fields */
   232→              <>
   233→                <div className="space-y-1.5">
   234→                  <Label htmlFor="name">{locale === "pt-BR" ? "Nome" : "Name"}</Label>
   235→                  <div className="relative">
   236→                    <User className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   237→                    <Input
   238→                      id="name"
   239→                      value={name}
   240→                      onChange={(e) => setName(e.target.value)}
   241→                      placeholder={locale === "pt-BR" ? "Seu nome" : "Your name"}
   242→                      className="pl-9"
   243→                    />
   244→                  </div>
   245→                </div>
   246→                <div className="space-y-1.5">
   247→                  <Label htmlFor="username">
   248→                    {locale === "pt-BR" ? "Nome de usuário" : "Username"}
   249→                  </Label>
   250→                  <div className="relative">
   251→                    <AtSign className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   252→                    <Input
   253→                      id="username"
   254→                      value={username}
   255→                      onChange={(e) => {
   256→                        const val = e.target.value.replace(/[^a-zA-Z0-9_]/g, "");
   257→                        setUsernameState(val);
   258→                        if (val.length >= 3) {
   259→                          checkUsername(val);
   260→                        } else {
   261→                          setUsernameStatus("invalid");
   262→                        }
   263→                      }}
   264→                      placeholder="username"
   265→                      className={cn(
   266→                        "pl-9",
   267→                        usernameStatus === "available" && "border-[var(--success)] focus-visible:ring-[var(--success)]",
   268→                        usernameStatus === "taken" && "border-[var(--destructive)] focus-visible:ring-[var(--destructive)]",
   269→                      )}
   270→                      maxLength={20}
   271→                    />
   272→                    {usernameStatus === "checking" && (
   273→                      <Loader2 className="absolute right-3 top-1/2 h-4 w-4 -translate-y-1/2 animate-spin text-muted-foreground" />
   274→                    )}
   275→                    {usernameStatus === "available" && (
   276→                      <span className="absolute right-3 top-1/2 -translate-y-1/2 text-xs font-medium text-[var(--success)]">
   277→                        {locale === "pt-BR" ? "Disponível" : "Available"}
   278→                      </span>
   279→                    )}
   280→                    {usernameStatus === "taken" && (
   281→                      <span className="absolute right-3 top-1/2 -translate-y-1/2 text-xs font-medium text-[var(--destructive)]">
   282→                        {locale === "pt-BR" ? "Em uso" : "Taken"}
   283→                      </span>
   284→                    )}
   285→                  </div>
   286→                  {usernameStatus === "invalid" && username.length > 0 && (
   287→                    <p className="text-xs text-muted-foreground">
   288→                      {locale === "pt-BR" ? "Mínimo 3 caracteres. Apenas letras, números e _." : "Min 3 characters. Letters, numbers and _ only."}
   289→                    </p>
   290→                  )}
   291→                </div>
   292→                <div className="space-y-1.5">
   293→                  <Label htmlFor="email">Email</Label>
   294→                  <div className="relative">
   295→                    <Mail className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   296→                    <Input
   297→                      id="email"
   298→                      type="email"
   299→                      value={email}
   300→                      onChange={(e) => setEmail(e.target.value)}
   301→                      placeholder="email@example.com"
   302→                      className="pl-9"
   303→                      required
   304→                    />
   305→                  </div>
   306→                </div>
   307→              </>
   308→            )}
   309→
   310→            {/* Password */}
   311→            <div className="space-y-1.5">
   312→              <Label htmlFor="password">{locale === "pt-BR" ? "Senha" : "Password"}</Label>
   313→              <div className="relative">
   314→                <Lock className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   315→                <Input
   316→                  id="password"
   317→                  type={showPassword ? "text" : "password"}
   318→                  value={password}
   319→                  onChange={(e) => setPassword(e.target.value)}
   320→                  placeholder="••••••••"
   321→                  className="pl-9 pr-9"
   322→                  required
   323→                  minLength={6}
   324→                />
   325→                <button
   326→                  type="button"
   327→                  onClick={() => setShowPassword(!showPassword)}
   328→                  className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   329→                >
   330→                  {showPassword ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
   331→                </button>
   332→              </div>
   333→            </div>
   334→
   335→            {/* Password confirmation (signup only) */}
   336→            {mode === "signup" && (
   337→              <div className="space-y-1.5">
   338→                <Label htmlFor="confirmPassword">
   339→                  {locale === "pt-BR" ? "Confirmar senha" : "Confirm password"}
   340→                </Label>
   341→                <div className="relative">
   342→                  <Lock className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   343→                  <Input
   344→                    id="confirmPassword"
   345→                    type={showConfirm ? "text" : "password"}
   346→                    value={confirmPassword}
   347→                    onChange={(e) => setConfirmPassword(e.target.value)}
   348→                    placeholder="••••••••"
   349→                    className={cn(
   350→                      "pl-9 pr-9",
   351→                      confirmPassword.length > 0 && password !== confirmPassword && "border-[var(--destructive)] focus-visible:ring-[var(--destructive)]",
   352→                      confirmPassword.length > 0 && password === confirmPassword && "border-[var(--success)] focus-visible:ring-[var(--success)]",
   353→                    )}
   354→                    required
   355→                    minLength={6}
   356→                  />
   357→                  <button
   358→                    type="button"
   359→                    onClick={() => setShowConfirm(!showConfirm)}
   360→                    className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   361→                  >
   362→                    {showConfirm ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
   363→                  </button>
   364→                </div>
   365→                {confirmPassword.length > 0 && password !== confirmPassword && (
   366→                  <p className="text-xs text-[var(--destructive)]">
   367→                    {locale === "pt-BR" ? "As senhas não coincidem." : "Passwords do not match."}
   368→                  </p>
   369→                )}
   370→              </div>
   371→            )}
   372→
   373→            <Button
   374→              type="submit"
   375→              className="w-full"
   376→              disabled={busy || !isSupabaseConfigured || (mode === "signup" && ((usernameStatus !== "available" && usernameStatus !== "idle" && usernameStatus !== "checking") || (confirmPassword.length > 0 && password !== confirmPassword)))}
   377→            >
   378→              {busy ? <Loader2 className="h-4 w-4 animate-spin" /> : mode === "signin"
   379→                ? (locale === "pt-BR" ? "Entrar" : "Sign In")
   380→                : (locale === "pt-BR" ? "Criar conta" : "Sign Up")}
   381→            </Button>
   382→          </form>
   383→
   384→          {/* Toggle mode */}
   385→          <p className="mt-4 text-center text-xs text-muted-foreground">
   386→            {mode === "signin"
   387→              ? (locale === "pt-BR" ? "Não tem conta? " : "No account? ")
   388→              : (locale === "pt-BR" ? "Já tem conta? " : "Already have an account? ")}
   389→            <button
   390→              onClick={() => {
   391→                setMode(mode === "signin" ? "signup" : "signin");
   392→                setUsernameStatus("idle");
   393→                setPassword("");
   394→                setConfirmPassword("");
   395→              }}
   396→              className="font-semibold text-primary hover:underline"
   397→            >
   398→              {mode === "signin"
   399→                ? (locale === "pt-BR" ? "Criar" : "Sign up")
   400→                : (locale === "pt-BR" ? "Entrar" : "Sign in")}
   401→            </button>
   402→          </p>
   403→
   404→          {/* Guest mode divider */}
   405→          <div className="relative my-4">
   406→            <div className="absolute inset-0 flex items-center">
   407→              <span className="w-full border-t border-border" />
   408→            </div>
   409→            <div className="relative flex justify-center">
   410→              <span className="bg-card px-2 text-[10px] uppercase tracking-wider text-muted-foreground/70">
   411→                {locale === "pt-BR" ? "ou" : "or"}
   412→              </span>
   413→            </div>
   414→          </div>
   415→
   416→          {/* Play as guest */}
   417→          <Button
   418→            type="button"
   419→            variant={isSupabaseConfigured ? "ghost" : "default"}
   420→            className={cn(
   421→              "w-full border border-dashed",
   422→              isSupabaseConfigured
   423→                ? "border-border text-muted-foreground hover:text-foreground hover:bg-muted/50"
   424→                : "border-primary/40 bg-primary/10 text-primary hover:bg-primary/20"
   425→            )}
   426→            onClick={enterGuestMode}
   427→          >
   428→            <Ghost className="h-4 w-4" />
   429→            {locale === "pt-BR" ? "Jogar como visitante" : "Play as guest"}
   430→          </Button>
   431→          <p className="mt-2 text-center text-[10px] text-muted-foreground/60">
   432→            {locale === "pt-BR"
   433→              ? "Explore o modo solo e o guia sem criar conta."
   434→              : "Explore solo mode and the guide without an account."}
   435→          </p>
   436→        </div>
   437→      </div>
   438→    </div>
   439→  );
   440→}

   442→function LoggedInHeader({ displayName, onSignOut, locale, isGuest, onLogin }: { displayName: string; onSignOut: () => void; locale: string; isGuest?: boolean; onLogin?: () => void }) {
   443→  const [showLogoutConfirm, setShowLogoutConfirm] = useState(false);
   444→
   445→  return (
   446→    <>
   447→      <div className="flex items-center gap-2">
   448→        <div className={cn(
   449→          "flex items-center gap-2 rounded-lg border border-border bg-card px-3 py-1.5",
   450→          isGuest && "border-dashed border-muted-foreground/40"
   451→        )}>
   452→          <div className={cn(
   453→            "flex h-7 w-7 items-center justify-center rounded-full text-xs font-bold",
   454→            isGuest ? "bg-muted text-muted-foreground" : "bg-primary/10 text-primary"
   455→          )}>
   456→            {isGuest ? <Ghost className="h-3.5 w-3.5" /> : displayName.charAt(0).toUpperCase()}
   457→          </div>
   458→          <span className="hidden text-sm font-medium sm:inline">{displayName}</span>
   459→          {isGuest && (
   460→            <span className="hidden rounded-full bg-muted px-1.5 py-0.5 text-[9px] font-semibold uppercase tracking-wider text-muted-foreground sm:inline">
   461→              {locale === "pt-BR" ? "Visitante" : "Guest"}
   462→            </span>
   463→          )}
   464→        </div>
   465→        {isGuest ? (
   466→          <Button
   467→            variant="outline"
   468→            size="sm"
   469→            onClick={onLogin}
   470→            className="gap-1.5"
   471→            aria-label={locale === "pt-BR" ? "Entrar" : "Sign in"}
   472→            title={locale === "pt-BR" ? "Entrar" : "Sign in"}
   473→          >
   474→            <LogIn className="h-4 w-4" />
   475→            <span className="hidden sm:inline">{locale === "pt-BR" ? "Entrar" : "Sign in"}</span>
   476→          </Button>
   477→        ) : (
   478→          <Button
   479→            variant="ghost"
   480→            size="icon"
   481→            onClick={() => setShowLogoutConfirm(true)}
   482→            aria-label="Sign out"
   483→            title="Sign out"
   484→          >
   485→            <LogOut className="h-4 w-4" />
   486→          </Button>
   487→        )}
   488→      </div>
   489→
   490→      {/* Logout confirmation popup */}
   491→      {showLogoutConfirm && (
   492→        <div
   493→          className="fixed inset-0 z-[300] flex items-center justify-center bg-background/80 backdrop-blur-sm"
   494→          onClick={() => setShowLogoutConfirm(false)}
   495→        >
   496→          <div
   497→            className="relative w-full max-w-sm rounded-xl border border-border bg-card p-6 shadow-2xl animate-float-up"
   498→            onClick={(e) => e.stopPropagation()}

   540→function GridNumbers() {
   541→  const [cells, setCells] = useState<GridCell[]>(() => generateGridCells());
   542→
   543→  useEffect(() => {
   544→    const interval = setInterval(() => {
   545→      setCells((prev) =>
   546→        prev.map((c) => {
   547→          if (c.phase === "hidden") {
   548→            if (Math.random() < 0.2) {
   549→              return {
   550→                ...c,
   551→                digit: Math.floor(Math.random() * 9) + 1,
   552→                phase: "in" as const,
   553→              };
   554→            }
   555→            return c;
   556→          }
   557→          if (c.phase === "in") {
   558→            if (c.opacity >= 0.12) return { ...c, phase: "hold" as const };
   559→            return { ...c, opacity: c.opacity + 0.02 };
   560→          }
   561→          if (c.phase === "hold") {
   562→            if (c.holdTime <= 0) return { ...c, phase: "out" as const };
   563→            return { ...c, holdTime: c.holdTime - 1 };
   564→          }
   565→          if (c.phase === "out") {
   566→            if (c.opacity <= 0) return { ...c, phase: "hidden" as const, opacity: 0 };
   567→            return { ...c, opacity: c.opacity - 0.02 };
   568→          }
   569→          return c;
   570→        })
   571→      );
   572→    }, 100);
   573→    return () => clearInterval(interval);
   574→  }, []);
   575→
   576→  return (
   577→    <div className="pointer-events-none absolute inset-0">
   578→      {cells.map((c) => (
   579→        <span
   580→          key={c.id}
   581→          className="absolute flex items-center justify-center font-bold tabular-nums text-muted-foreground transition-opacity duration-100"
   582→          style={{
   583→            left: `${c.col * 32}px`,
   584→            top: `${c.row * 32}px`,
   585→            width: "32px",
   586→            height: "32px",
   587→            fontSize: "1rem",
   588→            opacity: c.phase === "hidden" ? 0 : c.opacity,
   589→          }}
   590→        >
   591→          {c.phase !== "hidden" ? c.digit : ""}
   592→        </span>
   593→      ))}
   594→    </div>
   595→  );
   596→}
   597→
   598→interface GridCell {
   599→  id: number;
   600→  digit: number;
   601→  col: number;
   602→  row: number;
   603→  opacity: number;
   604→  phase: "hidden" | "in" | "hold" | "out";
   605→  holdTime: number;
   606→}
   607→
   608→function generateGridCells(): GridCell[] {
   609→  return Array.from({ length: 80 }, (_, i) => ({
   610→    id: i,
   611→    digit: Math.floor(Math.random() * 9) + 1,
   612→    col: Math.floor(Math.random() * 60),
   613→    row: Math.floor(Math.random() * 40),
   614→    opacity: 0,
   615→    phase: (["hidden", "hidden", "hidden", "in"] as const)[Math.floor(Math.random() * 4)],
   616→    holdTime: Math.floor(Math.random() * 40) + 10,
   617→  }));
   618→}
